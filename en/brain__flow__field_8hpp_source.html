37 bool build(map_type &map, const ISite &goal)
38 {
39 _map = &map;
40 _width = static_cast<int>(map.get_width());
41 _height = static_cast<int>(map.get_height());
42 const std::size_t total = map.size();
43 _cost.assign(total, unreachable);
44 _next.assign(total, -1);
45 _built = false;
46
47 if (!map.contains(goal) || map.has_lock(goal))
48 return false;
49 _goal = goal;
50
51 immune_system::cost_queue open;
52
53 const int goal_index = index_of(goal);
54 _cost[goal_index] = 0;
55 open.push({0, goal_index});
56
57 while (!open.empty())
58 {
59 immune_system::cost_entry e = open.top();
60 open.pop();
61 if (e.cost != _cost[e.id])
62 continue; // stale entry
63
64 ISite site {e.id % _width, e.id / _width};
65 for (int s = 0; s < map.identity.length; ++s)
66 {
67 ISite nb {site.x + map.identity.horizontal[s], site.y + map.identity.vertical[s]};
68 if (!map.contains(nb) || map.has_lock(nb) || !map._find_can_step(site, s))
69 continue;
70
71 // the step costs are symmetric, so walking back from the goal gives the cost toward it
72 weight_t cost = e.cost + map.identity.g_weight[s];
73 int nb_index = index_of(nb);
74 if (cost < _cost[nb_index])
75 {
76 _cost[nb_index] = cost;
77 _next[nb_index] = e.id;
78 open.push({cost, nb_index});
79 }
80 }
81 }
82
83 _built = true;
84 return true;
85 }

145 template <typename list_type>
146 bool path(const ISite &from, list_type &out) const
147 {
148 out.clear();
149 if (!reachable(from))
150 return false;
151 ISite current = from, following;
152 out.push_back(current);
153 while (next(current, following) && !(following.x == current.x && following.y == current.y))
154 {
155 current = following;
156 out.push_back(current);
157 }
158 return true;
159 }
